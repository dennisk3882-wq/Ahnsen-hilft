def rect(d,xy,fill,outline=None,w=1): d.rectangle(xy,fill=fill,outline=outline,width=w)
def line(d,xy,fill,w=1): d.line(xy,fill=fill,width=w)

def text(d,xy,s,fill=PAL['white'],anchor=None):
    d.text(xy,s,font=FONT,fill=fill,anchor=anchor)

def cloud(d,x,y,s=1):
    c=(235,240,224)
    for dx,dy,r in [(0,8,14),(18,2,18),(40,9,13),(24,12,18)]:
        d.ellipse((x+(dx-r)*s,y+(dy-r)*s,x+(dx+r)*s,y+(dy+r)*s),fill=c)

def tree(d,x,y,s=1):
    rect(d,(x-3*s,y,x+3*s,y+17*s),PAL['trunk'])
    for dx,dy,r,c in [(-7,-4,11,PAL['tree2']),(7,-5,12,PAL['tree2']),(0,-13,14,PAL['tree']),(0,0,12,PAL['tree'])]:
        d.ellipse((x+(dx-r)*s,y+(dy-r)*s,x+(dx+r)*s,y+(dy+r)*s),fill=c)
    for dx,dy in [(-6,-12),(5,-8),(1,-20)]: d.rectangle((x+dx*s,y+dy*s,x+(dx+2)*s,y+(dy+2)*s),fill=(123,166,65))

def bush(d,x,y,s=1):
    for dx in (-8,0,8): d.ellipse((x+(dx-7)*s,y-7*s,x+(dx+7)*s,y+7*s),fill=PAL['tree'])
    for dx in (-5,5): d.rectangle((x+dx*s,y-2*s,x+(dx+2)*s,y),fill=(226,154,84))

def road_h(d,y,h=55):
    rect(d,(0,y,W,y+h),PAL['road'])
    rect(d,(0,y,W,y+5),PAL['road2']); rect(d,(0,y+h-5,W,y+h),PAL['road2'])
    for x in range(8,W,34): rect(d,(x,y+h//2-2,x+18,y+h//2+2),PAL['line'])

def road_v(d,x,w=55,y0=0,y1=H):
    rect(d,(x,y0,x+w,y1),PAL['road'])
    rect(d,(x,y0,x+5,y1),PAL['road2']); rect(d,(x+w-5,y0,x+w,y1),PAL['road2'])
    for y in range(y0+8,y1,34): rect(d,(x+w//2-2,y,x+w//2+2,y+18),PAL['line'])

def sidewalk(d,xy): rect(d,xy,PAL['walk'],(118,119,115),2)

def crosswalk_h(d,x,y,w=55):
    for yy in range(y,y+50,8): rect(d,(x,yy,x+w,yy+4),PAL['white'])
def crosswalk_v(d,x,y,h=55):
    for xx in range(x,x+50,8): rect(d,(xx,y,xx+4,y+h),PAL['white'])

def house(d,x,y,w=74,h=60,roof=None,wall=None,stories=1,label=None):
    roof=roof or random.choice([PAL['roof'],(164,80,42),PAL['blue_roof']]); wall=wall or random.choice([PAL['wall'],(203,183,145),(193,175,136)])
    rect(d,(x,y,x+w,y+h),wall,(45,52,57),3)
    d.polygon([(x-5,y+3),(x+w//2,y-25),(x+w+5,y+3)],fill=roof,outline=(68,45,37))
    for i in range(0,w,10): line(d,(x+i,y-1,x+w//2,y-24),tuple(max(0,c-18) for c in roof),1)
    dw=14; rect(d,(x+w//2-dw//2,y+h-28,x+w//2+dw//2,y+h),(89,60,39),(40,40,40),2)
    rect(d,(x+w//2+4,y+h-15,x+w//2+6,y+h-13),PAL['gold'])
    wy=y+12
    for row in range(stories):
        yy=wy+row*22
        for xx in (x+12,x+w-27):
            rect(d,(xx,yy,xx+15,yy+13),PAL['window'],PAL['white'],2)
            line(d,(xx+7,yy,xx+7,yy+13),PAL['white']); line(d,(xx,yy+6,xx+15,yy+6),PAL['white'])
    if label:
        rect(d,(x+6,y+h-18,x+w-6,y+h-4),PAL['navy'])
        text(d,(x+w//2,y+h-11),label, PAL['gold'],anchor='mm')

def flowers(d,x,y,w,h,seed=1):
    rng=random.Random(seed)
    for _ in range(max(8,w*h//180)):
        xx=rng.randint(x,x+w); yy=rng.randint(y,y+h)
        c=rng.choice([(230,138,76),(232,205,71),(210,90,91),(205,154,208)])
        d.rectangle((xx,yy,xx+2,yy+2),fill=c)
